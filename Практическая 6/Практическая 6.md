public class CountdownService extends Service {
    private Thread worker;

    @Override
    public int onStartCommand(Intent intent, int flags, int startId) {
        worker = new Thread(() -> {
            for (int seconds = 60; seconds >= 0 &&
                    !Thread.currentThread().isInterrupted(); seconds--) {
                android.util.Log.d("CountdownService", "Осталось: " + seconds);
                try {
                    Thread.sleep(1000);
                } catch (InterruptedException e) {
                    Thread.currentThread().interrupt();
                    break;
                }
            }
            stopSelf(startId);
        });
        worker.start();
        return START_NOT_STICKY;
    }

    @Override
    public void onDestroy() {
        if (worker != null) worker.interrupt();
        super.onDestroy();
    }

    @Override
    public IBinder onBind(Intent intent) {
        return null;
    }
}

public Cursor query(Uri uri, String[] projection, String selection,
                    String[] selectionArgs, String sortOrder) {
    SQLiteDatabase db = helper.getReadableDatabase();
    int match = MATCHER.match(uri);
    if (match == ITEMS) {
        return db.query("notes", projection, selection, selectionArgs,
                null, null, sortOrder);

```java
@Override
public int bulkInsert(Uri uri, ContentValues[] values) {
    SQLiteDatabase db = helper.getWritableDatabase();
    int count = 0;
    db.beginTransaction();
    try {
        SQLiteStatement statement = db.compileStatement(
                "INSERT INTO products(name, price) VALUES(?, ?)");
        for (ContentValues value : values) {
            statement.clearBindings();
            statement.bindString(1, value.getAsString("name"));
            statement.bindDouble(2, value.getAsDouble("price"));
            statement.executeInsert();
            count++;
        }
        db.setTransactionSuccessful();
    } finally {
        db.endTransaction();
    }
    if (count > 0) getContext().getContentResolver().notifyChange(uri, null);
    return count;
}
```

@Override
public String[] getStreamTypes(Uri uri, String mimeTypeFilter) {
    String[] types = {"application/pdf", "text/csv"};
    List<String> result = new ArrayList<>();
    for (String type : types) {
        if (Intent.normalizeMimeType(mimeTypeFilter) == null ||
                type.matches(mimeTypeFilter.replace("*", ".*"))) {
            result.add(type);
        }
    }
    return result.toArray(new String[0]);
}

import android.os.Bundle;

public class MainActivity extends AppCompatActivity {

private void calculate(char operation) {
    String firstText = editTextFirst.getText().toString().trim();
    String secondText = editTextSecond.getText().toString().trim();

    if (firstText.isEmpty() || secondText.isEmpty()) {
        textViewResult.setText("Заполните оба поля");
        return;
    }

    try {
        double first = Double.parseDouble(firstText);
        double second = Double.parseDouble(secondText);
        double result;

        switch (operation) {
            case '+':
                result = first + second;
                break;

            case '-':
                result = first - second;
                break;

            case '*':
                result = first * second;
                break;

            case '/':
                if (second == 0) {
                    textViewResult.setText("На ноль делить нельзя");
                    return;
                }
                result = first / second;
                break;

            default:
                return;
        }

        textViewResult.setText(String.valueOf(result));

    } catch (NumberFormatException e) {
        textViewResult.setText("Ошибка ввода");
    }
}

## Задание 46. Калькулятор автокредита

**Задание:** Рассчитать ежемесячный платеж и показать на втором экране график выплат.

### Решение

Формула аннуитетного платежа:

```text
A = P * r * (1 + r)^n / ((1 + r)^n - 1)
```

где `P` — сумма кредита, `r` — месячная ставка, `n` — количество месяцев.

Java:

```java
buttonCalculate.setOnClickListener(v -> {
    try {
        double principal = Double.parseDouble(
                editTextAmount.getText().toString()
        );

        double annualRate = Double.parseDouble(
                editTextRate.getText().toString()
        );

        int months = Integer.parseInt(
                editTextMonths.getText().toString()
        );

        if (principal <= 0 ||
                annualRate < 0 ||
                months <= 0) {

            textViewResult.setText(
                    "Введите корректные значения"
            );

            return;
        }

        double monthlyRate =
                annualRate / 100.0 / 12.0;

        double payment;

        if (monthlyRate == 0) {
            payment = principal / months;
        } else {
            double factor = Math.pow(
                    1 + monthlyRate,
                    months
            );

            payment =
                    principal *
                    monthlyRate *
                    factor /
                    (factor - 1);
        }

        Intent intent = new Intent(
                MainActivity.this,
                ScheduleActivity.class
        );

        intent.putExtra("EXTRA_PAYMENT", payment);
        intent.putExtra("EXTRA_MONTHS", months);

        startActivity(intent);

    } catch (NumberFormatException e) {
        textViewResult.setText("Ошибка ввода");
    }
});
```
